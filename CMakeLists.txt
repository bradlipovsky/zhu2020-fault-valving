cmake_minimum_required(VERSION 3.12)
project(independent_fault_valving LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
find_path(FFTW_INCLUDE_DIR fftw3.h HINTS ${CMAKE_SOURCE_DIR}/.deps/include)
find_library(FFTW_LIBRARY fftw3 HINTS ${CMAKE_SOURCE_DIR}/.deps/lib)
if(NOT FFTW_INCLUDE_DIR OR NOT FFTW_LIBRARY)
  message(FATAL_ERROR "Install FFTW3 or run scripts/bootstrap.sh")
endif()
add_executable(valving src/model.cpp)
target_include_directories(valving PRIVATE ${FFTW_INCLUDE_DIR})
target_link_libraries(valving PRIVATE ${FFTW_LIBRARY} OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(valving PRIVATE stdc++fs)
endif()
target_compile_options(valving PRIVATE -O3 -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME equations_and_limits COMMAND valving --test)
add_executable(coupled_test tests/coupled.cpp)
target_include_directories(coupled_test PRIVATE ${FFTW_INCLUDE_DIR})
target_link_libraries(coupled_test PRIVATE ${FFTW_LIBRARY} OpenMP::OpenMP_CXX)
target_compile_options(coupled_test PRIVATE -O3 -Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(coupled_test PRIVATE stdc++fs)
endif()
add_test(NAME nonlinear_coupling COMMAND coupled_test)
