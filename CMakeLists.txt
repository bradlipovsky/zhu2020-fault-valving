cmake_minimum_required(VERSION 3.16)
project(fault_valving LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(FFTW_INCLUDE_DIR fftw3.h HINTS ${CMAKE_SOURCE_DIR}/build/fftw/include)
find_library(FFTW_LIBRARY fftw3 HINTS ${CMAKE_SOURCE_DIR}/build/fftw/lib)
if(NOT FFTW_INCLUDE_DIR OR NOT FFTW_LIBRARY)
  message(FATAL_ERROR "Install FFTW3 development files or run scripts/install_fftw.sh")
endif()
add_executable(valving src/valving.cpp)
target_include_directories(valving PRIVATE ${FFTW_INCLUDE_DIR})
target_link_libraries(valving PRIVATE ${FFTW_LIBRARY})
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
  target_link_libraries(valving PRIVATE stdc++fs)
endif()
target_compile_options(valving PRIVATE -O3 -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME physics COMMAND valving --test)
